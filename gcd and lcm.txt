package main

import "fmt"

func main() {
    var a, b, x, y int
    fmt.Scan(&a, &b)

    x = a
    y = b

    for y != 0 {
        x, y = y, x%y
    }

    gcd := x
    lcm := (a * b) / gcd

    fmt.Println("GCD =", gcd)
    fmt.Println("LCM =", lcm)
}

// Output:
// Input: 12 18
// GCD = 6
// LCM = 36
